```python
"""Deterministic legality repair: the model proposes, code disposes.

Repairs drafts against the code-owned trigger/target legality table for the
chassis in hand — repairs, never rejects, so the keyless path cannot fail.
Only a SELF chassis (a worn/used kit) constrains composition at v1; on every
other chassis all (kind, target) combinations are intended compositions
(lifedrain, the healing gun, self-costs). Semantics on a self-kit: ON_HIT =
"when you get hit" (needs a chassis that can absorb a hit — the `protective`
tag); ON_KILL = "when you down something while this is active" (the payoff
lands on you). The protective rule ships defined-but-dormant: the only
protective catalog item today (riot shield) locks GUARD.
"""
from __future__ import annotations

import copy

from resolver.contract import Delivery, GadgetDraft, Target, Trigger
from resolver.items import Build

# Nothing is thrown, placed, or blocked-with on a self-kit.
_SELF_ILLEGAL: frozenset[Trigger] = frozenset({
    Trigger.ON_IMPACT, Trigger.ON_TRIGGER, Trigger.ON_BLOCK,
})
# Stages whose effects can only land on the wielder.
_RETARGET_SELF: frozenset[Trigger] = frozenset({
    Trigger.ON_USE, Trigger.ON_EXPIRE, Trigger.ON_KILL,
})


def repair(draft: GadgetDraft, build: Build) -> GadgetDraft:
    if build.delivery is not Delivery.SELF:
        return copy.deepcopy(draft)
    protective = "protective" in build.delivery_item.tags
    out = copy.deepcopy(draft)
    for stage in out.stages:
        if stage.trigger in _SELF_ILLEGAL:
            stage.trigger = Trigger.ON_USE
        elif stage.trigger is Trigger.ON_HIT and not protective:
            stage.trigger = Trigger.ON_USE
        if stage.trigger in _RETARGET_SELF:
            for effect in stage.effects:
                effect.target = Target.SELF
    return out
```

```python
def _pairs(draft: GadgetDraft) -> set[tuple[EffectKind, Target]]:
    return {(e.kind, e.target) for stage in draft.stages for e in stage.effects}


def derive_harmless(draft: GadgetDraft) -> bool:
    """Harmless = can't hurt others: no lethal kind aimed at STRUCK.
    Hurting yourself is a cost, not a threat."""
    return not any(k in LETHAL_KINDS and t is Target.STRUCK for k, t in _pairs(draft))


def derive_category(draft: GadgetDraft) -> Category:
    pairs = _pairs(draft)
    if not pairs:
        return Category.DUD
    if any(k in LETHAL_KINDS and t is Target.STRUCK for k, t in pairs):
        return Category.WEAPON
    if any(k in CONTROL_KINDS and t is Target.STRUCK for k, t in pairs):
        return Category.CONTROL
    if any(k in SELF_KINDS or (t is Target.SELF and k not in UTILITY_KINDS)
           for k, t in pairs):
        return Category.SUPPORT
    if any(k in UTILITY_KINDS for k, _ in pairs):
        return Category.UTILITY
    return Category.TRINKET
```

- [ ] **Step 3: Implement in `resolver/clamp.py`.** Add `Effect, EffectKind` to the contract import (EffectKind already added in Task 3). Replace lines 28-67 (`OVER_BUDGET_FACTOR` through `fill_budget` — keep `TIER_BUDGET` as is) with:

```python
OVER_BUDGET_FACTOR: float = 1.12  # the ~10-15% over-budget cap
DUD_FLOOR: float = 1.0            # a DUD is pathetic, never literally nothing

# Exchange rates: what one unit of each field costs against the tier budget.
# Structure now, values later — deliberately crude placeholders (> 0 is the
# point: duration and count can no longer ride free on a full damage budget).
# Tuning any of these is a deterministic re-clamp, never a re-bake. W_AMOUNT
# is per-kind so the self-family rate (heal-80 vs damage-80) is a named knob.
W_AMOUNT: dict[EffectKind, float] = {k: 1.0 for k in EffectKind}
W_DURATION: float = 2.0
W_RADIUS: float = 0.5
W_COUNT: float = 3.0


def effect_cost(effect: Effect) -> float:
    return (effect.amount * W_AMOUNT[effect.kind]
            + effect.duration * W_DURATION
            + effect.radius * W_RADIUS
            + effect.count * W_COUNT)


def total_cost(draft: GadgetDraft) -> float:
    return sum(effect_cost(e) for stage in draft.stages for e in stage.effects)


def _scale(draft: GadgetDraft, factor: float) -> GadgetDraft:
    out = copy.deepcopy(draft)
    for stage in out.stages:
        for e in stage.effects:
            e.amount *= factor
            e.duration *= factor
            e.radius *= factor
    return out


def _count_cost(draft: GadgetDraft) -> float:
    return sum(e.count * W_COUNT for stage in draft.stages for e in stage.effects)


def _fit(draft: GadgetDraft, target_total: float) -> GadgetDraft:
    """Scale so total_cost lands on target_total: the float fields share one
    factor (proportions preserved); counts are integers with fixed cost and
    only floor-scale down when their cost alone exceeds the target (small
    under-budget residue from the integer floor is acceptable)."""
    out = copy.deepcopy(draft)
    counts = _count_cost(out)
    if counts > target_total:
        factor = target_total / counts
        for stage in out.stages:
            for e in stage.effects:
                e.count = int(e.count * factor)
        counts = _count_cost(out)
    floats = total_cost(out) - counts
    if floats <= 0:
        return out
    return _scale(out, (target_total - counts) / floats)


def clamp_magnitudes(draft: GadgetDraft, tier: Tier) -> GadgetDraft:
    total = total_cost(draft)
    if total == 0:
        return copy.deepcopy(draft)
    cap = TIER_BUDGET[tier] * OVER_BUDGET_FACTOR
    if total > cap:
        return _fit(draft, cap)
    if total < DUD_FLOOR:
        return _fit(draft, DUD_FLOOR)
    return copy.deepcopy(draft)


def fill_budget(draft: GadgetDraft, tier: Tier) -> GadgetDraft:
    """Scale a draft so its weighted cost lands exactly on the tier budget,
    preserving the composition's proportions. The LLM path calls this on
    parsed drafts (models free-hand magnitudes); the deterministic composer
    already fills the budget by construction. Zero-cost drafts pass through —
    there are no proportions to scale. Never called from `apply` — solo
    stays pathetic.
    """
    total = total_cost(draft)
    if total == 0:
        return copy.deepcopy(draft)
    return _fit(draft, TIER_BUDGET[tier])
```

````markdown
# Capability Contract — v1

The vocabulary the resolver composes over, and the wall the renderer must implement. **The model may only emit these tokens** — it can never ask for something the renderer can't build. Creativity lives in *composing* these freely (which primitives, in what sequence, on what triggers, at which target), not in inventing new ones.

Rule (from the POC, still true): **never add a token until the renderer can both execute AND present it.** The enums are the capability contract.

v1 = v0 tightened against the 2026-07-16 playtest evidence (see `docs/superpowers/specs/2026-07-16-contract-v1-design.md`): every effect declares who it hits, derivation keys on (kind, target), self-kit trigger semantics are law, and duration/radius/count cost budget. No enum family gained a token except `Target`, which is a validation axis (who an existing effect applies to — the renderer already had to know), not a renderer capability.

---

## Delivery — the chassis (how the weapon reaches the world)

Unchanged from v0. One per weapon, locked in code from the delivery-slot item.

Ranged/thrown: `HITSCAN`, `PROJECTILE`, `LOBBED`, `BEAM`, `THROWN`, `SPRAY`, `PLACED`.
Melee: `SWING`, `THRUST`, `SLAM`, `WHIP`, `GUARD` (enables `ON_BLOCK`).
Self/mobility: `SELF` (a kit worn or used on yourself — see trigger semantics below), `DASH`.

## EffectKind — the primitive payloads (what happens)

Unchanged from v0.

Damage family (lethal): `DAMAGE`, `BURN`, `EXPLODE`, `PIERCE`, `CHAIN` (+ `SUMMON` counts lethal).
Control family: `KNOCKBACK`, `LAUNCH`, `STAGGER`, `SNARE`, `SLOW`, `FREEZE`.
Self family: `HEAL`, `SHIELD`, `SPEED`.
Utility: `LURE`. (`SUMMON` lives here in flavor, in the damage family in law.)

## Target — who the effect applies to (NEW in v1)

- `STRUCK` — whatever the effect connects with: the thing hit, snared, or dosed.
- `SELF` — the wielder.

Required on **every** effect; there is no default (a baked scaffold can never contain a "whatever the default was that week" absence). Composition, not new tokens, expresses the interesting cases:

- Lifedrain = `DAMAGE@STRUCK + HEAL@SELF` on one stage — a weapon that feeds its user.
- The healing gun = `HEAL@STRUCK` — a dose fired into whatever the shot touches.
- A self-cost = `DAMAGE@SELF` riding a benefit — the adrenaline crash.

## Trigger — when a stage fires

Tokens unchanged from v0: `ON_USE`, `ON_HIT`, `ON_IMPACT`, `ON_TRIGGER`, `ON_EXPIRE`, `ON_KILL`, `ON_BLOCK`.

**Self-kit semantics (SELF chassis), enforced by deterministic repair (`resolver/validate.py`) — repaired, never rejected:**

| Rule | Repair |
| --- | --- |
| `ON_USE`, `ON_EXPIRE`, `ON_KILL` always legal | — |
| `ON_HIT` means "when *you* get hit" — legal only if the chassis item carries the `protective` tag | retrigger stage to `ON_USE` |
| `ON_IMPACT`, `ON_TRIGGER`, `ON_BLOCK` illegal (nothing is thrown, placed, or blocked-with) | retrigger stage to `ON_USE` |
| Effects in `ON_USE`/`ON_EXPIRE` stages target `SELF` | retarget to `SELF` |
| Effects in a legal reactive `ON_HIT` stage may target either (`STRUCK` = whoever hit you — retaliation) | — |
| Effects in `ON_KILL` stages target `SELF` (the reward reading; the target is dead) | retarget to `SELF` |

`ON_KILL` on a self-kit = "when you down something while this is active." The protective rule ships defined-but-dormant: the only `protective` item today (riot shield) locks a GUARD chassis; the first worn protective item activates it with zero code change.

**Non-SELF chassis: every (kind, target) combination is legal.** Lifedrain, the healing gun, and self-costs are intended compositions.

## Category & harmless — derived, never trusted (code re-computes both)

`harmless` = no lethal kind targeting `STRUCK`. Harmless means *can't hurt others*; hurting yourself is a cost, not a threat.

`derive_category` walks in priority order over the (kind, target) pairs present:

1. no effects → `DUD`
2. any lethal kind @`STRUCK` → `WEAPON`
3. any control kind @`STRUCK` → `CONTROL`
4. any self-family kind (either target) **or** any non-`LURE` kind @`SELF` → `SUPPORT`
5. any `LURE` → `UTILITY`
6. else → `TRINKET`

Accepted edge (documented, not special-cased): a build whose only effect is `DAMAGE@SELF` derives `SUPPORT`.

## Budget — weighted cost (NEW in v1)

Every effect field costs budget; raw amount-sum is dead:

```
cost(effect) = amount   * W_AMOUNT[kind]     # per-kind table, all 1.0 at v1
             + duration * W_DURATION
             + radius   * W_RADIUS
             + count    * W_COUNT
```

Weights live beside `TIER_BUDGET` in `resolver/clamp.py` — one tuning location, code-owned; values are deliberately crude placeholders until the renderer exists. Tuning them is a deterministic re-clamp, never a re-bake. Normalization scales the float fields by one shared factor (proportions preserved); counts are integers with fixed cost and floor-scale down only when count cost alone exceeds the budget. `target` does not affect cost at v1 (a self-cost *discount* is renderer-era tuning). Solo base-use stays pinned at 2.0 and is never budget-filled.

## Gadget schema (the output shape)

```
Effect:
  kind: EffectKind
  target: Target                # REQUIRED — who it applies to
  amount: float  (>= 0)
  duration: float
  radius: float
  count: int
  note: str

Stage:
  trigger: Trigger
  effects: list[Effect]
  note: str

GadgetDraft (model-facing):     name, description, logic, stages
ResolvedGadget (code-assembled): draft + delivery, tier, category, harmless, color, params
```

Params-by-delivery-family tables are unchanged from v0 (`resolver/items.py:params_for`).

## Ownership line — model vs code

- **Model owns:** meaning, name, `logic`, composition — which effects, sequence, triggers, targets, the interpretation of *these specific items*.
- **Code owns:** `delivery` (locked from item), `color`, tier/budget and all weights, caps/floors, legality repair, `harmless` and `category` (re-derived from (kind, target)), the ~10-15% over-budget cap. Model proposes; code disposes.

## v1 open questions

1. `LURE@SELF` (a taunt) stays legal on non-SELF chassis — maximally permissive where no invariant is at stake; revisit on live abuse.
2. Weight *values* are placeholders; play-tuned renderer-era, `TIER_BUDGET`-family knobs.
3. v0's open questions 1/3/4 (melee params, DASH/LAUNCH overlap, slice scope) carry forward unchanged.
````
